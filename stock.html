if (!categoryId) { table.innerHTML = ` Category hin argamne. `; return; } try { /* ================================================ LOAD ITEMS BY CATEGORY_ID ONLY ================================================ */ const itemsResponse = await fetch( SUPABASE_URL + "/rest/v1/items?category_id=eq." + encodeURIComponent(categoryId) + "&select=*", { headers: getHeaders() } );
